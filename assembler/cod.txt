; --- Test complet Assembler GP 16-bit ---
; Incarcare date initiale in registri
LDR X, 15        ; Incarca valoarea 15 in X
LDR Y, 5         ; Incarca valoarea 5 in Y

; Test instructiuni matematice (rutate catre ALU)
ADD X, 3         ; X = X + 3
SUB Y, 2         ; Y = Y - 2
MUL X, 2         ; X = X * 2
DIV X, 4         ; X = X / 4 (impartire intreaga)
MOD Y, 2         ; Y = Y % 2
INC X            ; Incrementare X
DEC Y            ; Decrementare Y

; Test instructiuni logice si shiftare
AND X, 1         ; X & 1 (Masking test)
OR Y, 8          ; Y | 8
XOR X, 3         ; X ^ 3
NOT X            ; Inversare biti X
LSL X, 2         ; Shift logic stanga cu 2 pozitii pe X
RSR Y, 1         ; Rotatie dreapta pe Y (1 bit)

; Test comparare si branch-uri
CMP X, 5         ; Compara X cu 5 (setari flags: Z, N, C, V)
BRZ 45           ; Salt la instructiunea de la adresa 45 daca Z=1
BRN 50           ; Salt la 50 daca Negative
BRO 60           ; Salt la 60 daca Overflow

; Test proceduri
JMP 100          ; Call procedura aflata la adresa 100 (adauga PC pe stiva)

; Testare tratare erori asamblor (aceste linii ar trebui sa fie raportate in consola)
ADD Z, 5
BAD_OPCODE X, 1

; Sfarsit apel (Return)
RET              ; Extrage adresa din stiva in PC
